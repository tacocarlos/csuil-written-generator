% --- Define the new command \questionBlock ---
% It takes 7 arguments:
% #1: Question number
% #2: The main question text
% #3: Option A
% #4: Option B
% #5: Option C
% #6: Option D
% #7: Option E
\newcommand{\questionBlock}[7]{%
    \par\noindent % Ensure we start on a new paragraph
    % Create the black box with white, bold text for the question number
    \begin{tcolorbox}[
        colback=white,       % Background color
        coltext=black,       % Text color
        fontupper=\bfseries, % Bold text
        sharp corners,       % No rounded corners
        boxrule=1pt,         % No border line
        width=5cm,           % Fixed width for the box
        left=4pt, right=4pt, top=2pt, bottom=2pt, % Internal padding
        halign=left          % Align text to the left
    ]
        Question #1
    \end{tcolorbox}
    \vspace{-0.3em} % Add a small negative vertical space to pull the text up

    % Display the main question text
    \noindent #2

    \vspace{1em} % Vertical space between the question and the options

    % Display the five options in a horizontal row using a tabular environment
    \noindent
    \begin{tabular}{@{}p{0.19\linewidth}p{0.19\linewidth}p{0.19\linewidth}p{0.19\linewidth}p{0.19\linewidth}@{}}
        \textbf{A)} #3 & \textbf{B)} #4 & \textbf{C)} #5 & \textbf{D)} #6 & \textbf{E)} #7
    \end{tabular}
    \vspace{2em} % Vertical space after the entire question block
    \iftoggle{separatePages}{\newpage}
}
